SQL de 1 Tabla para tu Proyecto de Supabase (Solo se ejecuta 1 vez) 
create table if not exists public.smartpiggy_families (
  family_code text primary key,
  payload jsonb not null,
  updated_at timestamptz not null default now()
);
alter table public.smartpiggy_families enable row level security;
create policy "smartpiggy_sync_all" on public.smartpiggy_families for all to anon, authenticated using (true) with check (true);
alter publication supabase_realtime add table public.smartpiggy_families;

En Android (Chrome / Edge / Samsung): 1. Toca los 3 puntos (⋮) arriba a la derecha.
